cmake_minimum_required(VERSION 3.14)
project(ScaleSimRamulator2 LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(RAMULATOR_ROOT "${PROJECT_ROOT}/ramulator2" CACHE PATH "Ramulator2 source directory")
find_library(RAMULATOR_LIBRARY ramulator PATHS "${RAMULATOR_ROOT}" NO_DEFAULT_PATH)
if(NOT RAMULATOR_LIBRARY)
  message(FATAL_ERROR "Build ramulator2/libramulator.so with ./setup.sh first")
endif()
add_executable(ramulator2-replay replay.cpp)
target_include_directories(ramulator2-replay PRIVATE "${RAMULATOR_ROOT}/src" "${RAMULATOR_ROOT}/ext/fmt/include")
target_link_libraries(ramulator2-replay PRIVATE "${RAMULATOR_LIBRARY}")

add_executable(ramulator2-online online.cpp)
target_include_directories(ramulator2-online PRIVATE "${RAMULATOR_ROOT}/src" "${RAMULATOR_ROOT}/ext/fmt/include")
target_link_libraries(ramulator2-online PRIVATE "${RAMULATOR_LIBRARY}")
